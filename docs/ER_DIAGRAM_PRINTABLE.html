    APP_USER {
        int user_id PK "Surrogate Key"
        string email UK "Unique Credential"
        string password_hash "Bcrypt Hash"
        string role "student | recruiter | admin"
        boolean is_active "Default TRUE"
    }

    STUDENT {
        int student_id PK "Surrogate Key"
        int user_id FK "1:1 with app_user"
        string usn UK "University Seat Number"
        string full_name "Legal Name"
        int dept_id FK "Enrolled Department"
        decimal cgpa "Academic GPA (0-10)"
        int year_of_study "1-4"
        int grad_year "Graduation Year"
        vector profile_embedding "384-d pgvector"
    }

    DEPARTMENT {
        int dept_id PK "Surrogate Key"
        string code UK "e.g. CSE, AIML, ECE"
        string name "Department Name"
    }

    COMPANY {
        int company_id PK "Surrogate Key"
        string name UK "Company Legal Name"
        string website "Corporate URL"
        string industry "Sector / Domain"
        string size_range "Employee Scale"
        boolean verified "Admin Approved"
    }

    RECRUITER {
        int recruiter_id PK "Surrogate Key"
        int user_id FK "1:1 with app_user"
        int company_id FK "Employer Company"
        string full_name "Contact Person"
        string designation "Job Title"
        string phone "Phone Number"
    }

    SKILL {
        int skill_id PK "Surrogate Key"
        string name UK "Canonical Name"
        string category "Domain Category"
    }

    INTERNSHIP_POSTING {
        int posting_id PK "Surrogate Key"
        int company_id FK "Hiring Company"
        int created_by FK "Recruiter ID"
        string title "Position Title"
        decimal stipend "Monthly Stipend"
        int openings "Available Seats"
        int filled "Filled Openings"
        decimal min_cgpa "Eligibility Cutoff"
        timestamp deadline "Application Cutoff"
        string status "draft | open | closed"
        vector embedding "384-d pgvector"
    }

    APPLICATION {
        int application_id PK "Surrogate Key"
        int student_id FK "Candidate"
        int posting_id FK "Internship"
        string current_stage "State Machine Enum"
        decimal match_score "Skill/Vector Match"
        timestamp applied_at "Submission Time"
    }

    APPLICATION_STAGE_HISTORY {
        int history_id PK "[WEAK ENTITY]"
        int application_id FK "Identifying Owner"
        string from_stage "Previous Stage"
        string to_stage "Next Stage"
        int changed_by FK "Authorized User"
        text reason "Audit Notes"
        timestamp changed_at "Discriminator"
    }

    INTERVIEW {
        int interview_id PK "[WEAK ENTITY]"
        int application_id FK "Identifying Owner"
        int round_no "Partial Discriminator"
        timestamp scheduled_at "Meeting Start"
        int duration_min "Length in Minutes"
        string mode "online | in-person"
        string status "scheduled | completed"
    }

    OFFER {
        int offer_id PK "[DEPENDENT ENTITY]"
        int application_id FK, UK "1:1 Identifying Owner"
        decimal stipend "Offered Amount"
        date start_date "Internship Start"
        date expires_at "Acceptance Deadline"
        string status "pending | accepted | declined"
    }

    STUDENT_SKILL {
        int student_id PK, FK "Composite Key"
        int skill_id PK, FK "Composite Key"
        int proficiency_level "Level 1 to 5"
    }

    POSTING_SKILL {
        int posting_id PK, FK "Composite Key"
        int skill_id PK, FK "Composite Key"
        int min_level "Minimum Level 1-5"
    }

    POSTING_ELIGIBLE_DEPT {
        int posting_id PK, FK "Composite Key"
        int dept_id PK, FK "Composite Key"
    }

    INTERVIEW_PANELIST {
        int interview_id PK, FK "Composite Key"
        int recruiter_id PK, FK "Composite Key"
    }

    POLICY_CONFIG {
        int policy_id PK "Surrogate Key"
        string policy_key UK "Config Key"
        string policy_value "Config Value"
        string description "Policy Scope"
    }

    AUDIT_LOG {
        int log_id PK "Surrogate Key"
        int user_id FK "Actor"
        string action "Event Type"
        string entity_name "Target Table"
        int entity_id "Target ID"
        timestamp created_at "Immutable Timestamp"
    }

Strong Entities (Independent)

  • app_user: Root identity for student, recruiter, admin.
  • department: PES University academic branches.
  • company: Corporate hiring partners.
  • skill: Master dictionary of technical skills.
  • policy_config & audit_log: Governance and security.

Weak & Dependent Entities

  • application_stage_history: Existence-dependent on application (Owner). Cascades on deletion; immutable.
  • interview: Subordinate to application; identified by (application_id, round_no).
  • offer: Dependent 1:1 on application.

Associative Entities (M:N)

  • application: Student ➔ Internship Posting
  • student_skill: Student ➔ Skill
  • posting_skill: Internship Posting ➔ Skill
  • posting_eligible_dept: Posting ➔ Dept
  • interview_panelist: Interview ➔ Recruiter
